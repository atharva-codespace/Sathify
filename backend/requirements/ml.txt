# Computer-vision dependencies for Module 3 (OCR) and Module 6 (face verification).
#
# INSTALLED SEPARATELY AND DELIBERATELY. This set pulls in PaddlePaddle,
# TensorFlow and PyTorch — several GB on disk and well over 512 MB of RSS at
# runtime. It does NOT fit on Render's free tier (512 MB RAM); see
# docs/free-tier-constraints.md for the deployment split we use instead.
#
#   python -m pip install -r requirements/ml.txt
#
# Verified available as cp313 Windows wheels on 2026-07-30.

# Brings in base.txt plus OpenCV and NumPy. Those two are pinned in cv.txt
# rather than here so CI can install the cheap half of the CV stack on its own.
-r cv.txt

# --- OCR: PaddleOCR primary -------------------------------------------------
paddlepaddle==3.1.1
paddleocr==3.1.1

# paddleocr 3.1.1 declares `paddlex>=3.1.0` with **no upper bound**, so a fresh
# install resolves whatever is newest — and 3.7.2 changed a constructor
# signature paddleocr still calls positionally:
#
#     TypeError: PaddlePredictorOption.__init__() takes 1 positional argument
#                but 2 were given
#
# The damage is quiet rather than loud. `PaddleOcrEngine.is_available()` only
# checks that the module imports, so the engine reports itself ready, fails at
# runtime on the first document, and `recognise()` silently falls through to
# EasyOCR — exactly as designed. Every scan then gets served by the fallback,
# accuracy drops to the fallback's, and nothing anywhere says so.
#
# UNVERIFIED: the bound below is the compatible minor line by inspection, not by
# a clean reinstall. Confirm with `pip install -r requirements/ml.txt` in a fresh
# environment and check `PaddleOcrEngine().run(...)` actually returns before
# trusting it. See docs/manual-test-cases.md, OCR-07.
paddlex>=3.1.0,<3.2

# --- OCR: EasyOCR automatic fallback ----------------------------------------
easyocr==1.7.2
# torch is pulled in by easyocr; pinned here so the CPU build is explicit.
torch==2.7.1
torchvision==0.22.1

# --- Face verification (Module 6) -------------------------------------------
deepface==0.0.100
tf-keras==2.20.1      # DeepFace requires the standalone Keras 2 shim
tensorflow==2.20.0    # first release with cp313 wheels

# --- PDF input support (OCR Stage 1) ----------------------------------------
pdf2image==1.17.0     # NOTE: also needs the poppler binary on PATH
